<script>
// 2026-08-21 — shared utility-nav logic, identical on every page (see the
// .utility-nav CSS comment for why). Reads a single sessionStorage pointer
// (verilume_active_account) that login.html / assessment.html's Save &
// Schedule flow set on successful sign-in, and that this same script's own
// Logout handler (and every page's own logout(), where one exists) clears.
// Deliberately self-contained (its own vlUtilBackendUrl(), not the page's
// BACKEND_URL) so it works unmodified on pages that don't otherwise talk to
// the backend at all (marketing pages, terms.html, etc.).
(function(){
  function vlUtilBackendUrl(){
    return (typeof window !== 'undefined' && window.location && window.location.protocol !== 'file:')
      ? ''
      : 'http://localhost:8787';
  }
  function vlActiveAccountId(){
    try { return localStorage.getItem('verilume_active_account') || ''; } catch (e) { return ''; }
  }
  function vlUtilityNavInit(){
    var accountId = vlActiveAccountId();
    var portalLink = document.getElementById('utilNavPortalLink');
    var authLink = document.getElementById('utilNavAuthLink');
    if (!portalLink || !authLink) return;
    if (accountId){
      // 2026-08-21 — per direct feedback, showing "Access My Portal" while
      // already ON portal.html is redundant (you're already there); hide
      // it there specifically, still show it everywhere else (marketing
      // pages, assessment.html, account.html, etc.) so there's always a
      // way back to the Portal from outside it.
      var onPortalPage = /\/?portal\.html$/.test((window.location && window.location.pathname) || '');
      if (onPortalPage){
        portalLink.style.display = 'none';
      } else {
        portalLink.href = 'portal.html?accountId=' + encodeURIComponent(accountId);
        portalLink.style.display = '';
      }
      authLink.textContent = 'Logout';
      authLink.href = '#';
      authLink.onclick = function(e){
        e.preventDefault();
        var token = null;
        try { token = localStorage.getItem('verilume_session_' + accountId); } catch (e2) {}
        var finish = function(){
          try {
            localStorage.removeItem('verilume_session_' + accountId);
            localStorage.removeItem('verilume_member_' + accountId);
            localStorage.removeItem('verilume_active_account');
          } catch (e3) {}
          location.href = '/';
        };
        if (token){
          fetch(vlUtilBackendUrl() + '/api/auth/logout', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ token: token })
          }).then(finish).catch(finish);
        } else {
          finish();
        }
      };
    } else {
      portalLink.style.display = 'none';
      authLink.textContent = 'Log in';
      authLink.href = 'login.html';
      authLink.onclick = null;
    }
  }
  if (document.readyState === 'loading'){
    document.addEventListener('DOMContentLoaded', vlUtilityNavInit);
  } else {
    vlUtilityNavInit();
  }
  // 2026-08-24, per direct report ("Access My Portal takes me to the
  // wrong account, even after logging in as the right one") — this only
  // ever ran once, on initial load. Two real ways that goes stale: (1)
  // the browser restores this exact page from bfcache on Back/Forward
  // navigation, which does NOT re-fire DOMContentLoaded, so the link's
  // href keeps whatever account (or none) was active when this page was
  // first painted, even if a login happened after that in this same tab
  // via a different page; (2) this page was left open in one tab while
  // login happened in another tab/window — localStorage changed, but
  // this tab's DOM never re-read it. 'pageshow' fires on both a normal
  // load AND a bfcache restore (check event.persisted to tell them
  // apart, though re-running is harmless either way); the 'storage'
  // event fires in OTHER tabs of the same origin when localStorage
  // changes in one tab, so both close the gap.
  window.addEventListener('pageshow', function(e){ vlUtilityNavInit(); });
  window.addEventListener('storage', function(e){
    if (!e.key || e.key === 'verilume_active_account') vlUtilityNavInit();
  });
  window.vlUtilityNavInit = vlUtilityNavInit; // exposed so login()/logout() elsewhere can re-run it after changing session state without a full reload
})();
</script>

// email and updates the verify step's copy — same endpoint/table
// assessment.html's Save Assessment modal uses for its phone/email

async function requestOnboardingVerification(){
  const introEl = document.getElementById('verifyIntro');
  const statusEl = document.getElementById('verifyStatus');
  statusEl.textContent = '';
  introEl.textContent = 'Sending a verification code…';
  try {
    const resp = await fetch(`${BACKEND_URL}/api/auth/request-signup-verification`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ email: state.adminEmail })
    });
    const data = await resp.json().catch(() => ({}));
    if (!resp.ok) throw new Error(data.error || `responded ${resp.status}`);
    state.verificationId = data.verificationId;
    introEl.textContent = `We emailed a 6-digit code to ${data.target}. Enter it below — this confirms you're a real person before your account is created.`;

    if (data.interimCode){

    introEl.textContent = 'Enter the verification code sent to your email.';
    statusEl.textContent = `Couldn't send a verification code right now. ${e.message || 'Try Resend in a moment.'}`;

// Round 83 (Deliverable #4) — real path: POST this session's real Company
// name + region to the backend, which creates a genuine `accounts` row and

async function createRealAccount(){
  const h = state.assessmentHandoff;
  const resp = await fetch(`${BACKEND_URL}/api/accounts`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      company: state.company || 'Your Company',
      industry: (h && h.industry) || '',
      footprint: state.region || '',
      url: (h && h.url) || '',
      description: (h && h.description) || '',
      productThemes: (h && h.productThemes) || '',
      audience: (h && h.audience) || [],
      wealth: (h && h.wealth) || [],
      assessedStages: (h && h.assessedStages) || [],
      cells: (h && h.cells) || [],
      tier: state.tier || undefined
    })
  });
  if (!resp.ok) throw new Error(`Backend responded ${resp.status}`);
  return resp.json(); // { accountId, accessCode, token, expiresAt }
}

// Added 2026-08-18 (registration rebuild) — creates this account's real
// admin/CMO user right after the account itself exists, per direct
// instruction: a real, individually memorable username (email) + a
// system-issued temp password, not the account-level access code above.
// Called from runProvisioning() below, right after createRealAccount()
// succeeds — this account's accountId has to exist first since
// register-admin 404s on an unknown one.
async function registerAdmin(accountId){
  const resp = await fetch(`${BACKEND_URL}/api/auth/register-admin`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      accountId,
      name: state.adminName || 'Account Admin',
      email: state.adminEmail,
      // 2026-08-20 — server.js now requires and consumes this: a verified,
      // unconsumed signup verification (see the 'verify' step above) is
      // what makes this an "actual account" per direct instruction. A
      // missing/expired/reused verificationId 400s here rather than
      // silently creating an unverified login.
      verificationId: state.verificationId,
      // Only present if the optional NDA step was actually visited and
      // signed — server logs this as a separate 'nda' legal-acceptance
      // row (method 'signature') alongside the always-present T&Cs/pricing
      // disclosure rows (method 'assumed-on-save') it writes regardless.
      nda: state.ndaSignedAt ? { signedAt: state.ndaSignedAt, signatureName: state.ndaSignatureName } : undefined
    })
  });
  if (!resp.ok){
    const data = await resp.json().catch(() => ({}));
    throw new Error(data.error || `register-admin responded ${resp.status}`);
  }
  return resp.json(); // { memberId, email, tempPassword, mustChangePassword, token, expiresAt }
}

function runProvisioning(){
  const items = document.querySelectorAll('.provision-item');
  items.forEach((item, i) => {
    setTimeout(() => { item.classList.add('done'); }, 500 * (i + 1));
  });
  const totalDelay = 500 * items.length;
  setTimeout(async () => {
    const note = document.getElementById('provisionNote');
    try {
      const { accountId, token } = await createRealAccount();
      state.accountId = accountId;
      persistSessionToken(accountId, token);
      document.getElementById('generatedAccountId').textContent = accountId;
      document.getElementById('accountIdCard').style.display = 'block';
      document.getElementById('provisionActions').style.display = 'flex';
      document.getElementById('manageAccountLink').href = `account.html?accountId=${encodeURIComponent(accountId)}`;
      // Registration rebuild (2026-08-18): the account-level access code
      // above is no longer what a real person signs in with — register
      // the real admin/CMO user (name/email collected at Step 1) and show
      // their temp password instead. If this call fails for any reason
      // (duplicate email, adminEmail missing from an old/resumed session,
      // etc.), account creation itself still succeeded above — this just
      // falls back to noting that the admin login couldn't be set up, so
      // the person isn't left thinking the whole step failed.
      try {
        const admin = await registerAdmin(accountId);
        persistSessionToken(accountId, admin.token);
        document.getElementById('generatedAccessCode').textContent = admin.tempPassword;
        document.getElementById('generatedAdminEmail').textContent = admin.email;
        document.getElementById('accessCodeBlock').style.display = 'block';
        note.classList.add('signed');
        note.textContent = `✓ This is a real account record in the local demo backend — "Manage your account" below will open already signed in as ${admin.email}.`;
      } catch (adminErr){
        console.warn('registerAdmin failed — account exists, but no admin login was created.', adminErr);
        document.getElementById('accessCodeBlock').style.display = 'none';
        note.classList.add('signed');
        note.textContent = `✓ Account created. Couldn't set up your admin login automatically (${adminErr && adminErr.message ? adminErr.message : (state.adminEmail ? 'that email may already be registered' : 'no email was captured')}) — go back and verify again, or contact Verilume support.`;
      }
    } catch (e){
      // Honest fallback, same convention as every other real-backend page in
      // this build: the backend wasn't reachable, so this session gets a
      // browser-only placeholder ID instead of pretending it's a real
      // account. Clearly labeled as such rather than looking identical to
      // the real path above.
      console.warn('createRealAccount: backend not reachable, falling back to a session-only placeholder ID.', e);
      const year = 2026; // session clock; the real path above stamps the actual create date
      const seq = String(Math.floor(100 + Math.random() * 900));
      const id = `CXM-${regionCode(state.region)}-${year}-${seq}`;
      state.accountId = id;
      document.getElementById('generatedAccountId').textContent = id;
      document.getElementById('accessCodeBlock').style.display = 'none';
      document.getElementById('accountIdCard').style.display = 'block';
      document.getElementById('provisionActions').style.display = 'flex';
      document.getElementById('manageAccountLink').href = `account.html?accountId=${encodeURIComponent(id)}`;
      note.textContent = `The local demo backend wasn't reachable at ${BACKEND_URL}, so this Account ID is a session-only placeholder, not a real saved account.`;
    }
  }, totalDelay);
}
